🔧 CUSTOMIZATION

[RECOMMENDATION]
  • COLLABORATIVE_WEIGHT = 0.6     # Collab filtering weight
  • CONTENT_WEIGHT = 0.4           # Content-based weight
  • MIN_SIMILARITY_THRESHOLD = 0.3 # Minimum similarity

  • FRESHNESS_WEIGHT = 0.3         # Freshness boost
  • NOVELTY_WEIGHT = 0.2           # Novelty boost
  • SERENDIPITY_WEIGHT = 0.1       # Surprise factor

[EVALUATION]
  • METRICS_K = 10                 # Evaluation at K items

[UI]
  • ITEMS_PER_PAGE = 10            # Results per page
  • THEME_PRIMARY_COLOR = #00D9FF  # Primary color
  • DARK_BG_COLOR = #0a0e27        # Background
